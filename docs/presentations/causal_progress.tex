\begin{frame}{总体流程}
  \begin{center}
    \small
    5{,}048 个站点 \ $\rightarrow$ \ 控制设计（同城匹配、GSC、MC）
    \ $\rightarrow$ \ 逐任务响应标签 \\
    $\rightarrow$ \ Response Artifact 发布
    \ $\rightarrow$ \ 四族事件研究验证
    \ $\rightarrow$ \ 逐标签可靠性权重
    \ $\rightarrow$ \ 表示学习监督
  \end{center}
  \vspace{0.5em}
  \begin{itemize}
    \item 每个任务 = 站点 $\times$ 结果族 $\times$ 方法 $\times$ donor scope
    \item 标签与诊断（方法、标准误、处理前拟合）一起保存，全链路可溯源
    \item 事件研究是标签质量的聚合验证；标签本身来自逐任务估计
  \end{itemize}
\end{frame}

\begin{frame}{因果对象与潜在结果}
  \begin{itemize}
    \item 一致性：$Y_{ikt}=D_{it}Y_{ikt}(1)+(1-D_{it})Y_{ikt}(0)$
    \item 动态处理效应：$\tau_{ikh}=Y_{ik,G_i+h}(1)-Y_{ik,G_i+h}(0)$
    \item 响应标签：$L_{ikh}=Y_{ik,G_i+h}-\hat Y_{ik,G_i+h}(0)$
    \item 队列动态 ATT：
      $ATT_k(h)=E\!\left[Y_{ik,G_i+h}(1)-Y_{ik,G_i+h}(0)\mid T_i=1\right]$
  \end{itemize}
  \vspace{0.5em}
  \begin{itemize}
    \item $h=0$ 是开通期冲击（保留真实观测），$h\ge 1$ 为之后的完整期
    \item 逐网格标签用于表示学习；队列 ATT 用于计量报告，两者分开
  \end{itemize}
\end{frame}

\begin{frame}{识别假设}
  \begin{itemize}
    \item \textbf{无提前反应}：清洁处理前窗口内 $Y(1)=Y(0)$（开通前 6 个月为预期缓冲）
    \item \textbf{匹配路径的条件平行趋势}：
      \[
        E[\Delta Y_{i,t}(0)\mid T_i=1, X_i=x]
        = E[\Delta Y_{j,t}(0)\mid C_j=1, X_j=x]
      \]
    \item \textbf{donor 未受处理与空间排除}：站点 1\,km 内网格不进入风险集
    \item \textbf{共同支持}：逐特征区间覆盖；Mahalanobis 距离与 holdout 误差描述高维支持
    \item \textbf{GSC 交互固定效应结构}：
      $Y_{it}(0)=\alpha_i+\delta_t+\lambda_i'F_t+\varepsilon_{it}$，载荷稳定
    \item \textbf{MC 低秩结构}：$Y(0)=L+E$，缺失模式保留处理前与 donor 支持
  \end{itemize}
\end{frame}

\begin{frame}{估计器（1）：匹配与 Abadie--Imbens ATT}
  \begin{itemize}
    \item Mahalanobis 距离
      $d_{ij}=\sqrt{(X_i-X_j)'\hat\Sigma^{-1}(X_i-X_j)}$
    \item 控制设计：M=5 候选 $\rightarrow$ 静态特征精炼 $\rightarrow$ 冻结控制身份
    \item 质量门禁：训练距离、holdout $RMS$、$MAX$ 均不超过 placebo 的 95\% 分位
    \item 月度标签（$B$ 为开通前 12 个清洁月均值）：
      \[
        L_{ikh}=(Y_{i,h}-B_i)-(Y_{j(i),h}-B_{j(i)})
      \]
    \item 完整趋势 ATT（Abadie--Imbens 偏差修正）：
      \[
        ATT=\frac{1}{N_T}\sum_i
        \left[\Delta Y_i-\Delta Y_{j(i)}+(X_i-X_{j(i)})'\hat\beta\right]
      \]
  \end{itemize}
\end{frame}

\begin{frame}{估计器（2）：广义合成控制（Xu GSC）}
  \begin{itemize}
    \item 未处理结果由单位效应、时间效应与低维共同因子决定：
      $Y_{it}(0)=\alpha_i+\delta_t+\lambda_i'F_t+\varepsilon_{it}$
    \item 目标拟合：$Y_{i,t}=a_i+F_t'b_i+\varepsilon_{i,t}$，
      $\hat Y_{i,t}(0)=\hat a_i+\hat F_t'\hat b_i$
    \item 因子数：rolling CV，候选 $r=0,\dots,5$（MSPE + 1SE 规则）
    \item 不确定性：200 次参数 bootstrap，$SE_t=sd(\tau_t^{(b)})$，正态区间
    \item 跨城版本用城市内 median/MAD 标准化，再在目标尺度上反标准化
  \end{itemize}
\end{frame}

\begin{frame}{估计器（3）：矩阵补全（MC）}
  \begin{itemize}
    \item 核范数目标：
      \[
        \min_L\ \frac{1}{|\Omega|}\sum_{(i,t)\in\Omega}(Y_{it}-L_{it})^2
        +\lambda\|L\|_*
      \]
    \item 两向固定效应初始化 + 奇异值软阈值
      $s_l^{\text{shrunk}}=\max(s_l-\lambda TN,\,0)$
    \item $\lambda$：rolling CV（20 折、1SE 规则），$\lambda=0$ 为合法端点
    \item 单位 jackknife 推断：$N_t$ 为事件期有效重拟合次数
      \[
        p_t^{(-j)}=N_t\hat\tau_t-(N_t-1)\hat\tau_t^{(-j)},\qquad
        SE_t^{JK}=\sqrt{\operatorname{var}(p_t^{(-j)})/N_t}
      \]
  \end{itemize}
\end{frame}

\begin{frame}{事件研究与平行趋势}
  \begin{itemize}
    \item TWFE 事件研究（匹配样本）：
      \[
        Y_{it}=\alpha_i+\gamma_t+\sum_{h\neq h_0}\beta_h
        \left[Treated_i\cdot \mathbf{1}(\text{event\_time}=h)\right]+\varepsilon_{it}
      \]
    \item 参考期 $h_0=-1$；$t=0$ 为正式冲击期；干净 lead 只用 $t<0$
    \item 预注册规范窗口：月度族 $\pm 24$；年度族 $[-6,\,+3]$
    \item 平行趋势联合检验：参考期前 5 个 clean lead（$-6..-2$）
      \[
        F=q^{-1}\hat\beta_{pre}'\hat V_{pre}^{-1}\hat\beta_{pre}
        \sim F(q,\,G-1)
      \]
    \item 聚类：CRV1（grid / city）；主解释使用城市聚类
  \end{itemize}
\end{frame}

\begin{frame}{生产与发布}
  \begin{itemize}
    \item 5{,}048 个站点；20{,}192 个 family 级任务全部终态
    \item 终态队列：9{,}162 skipped；7{,}553 matched；2{,}322 GSC；1{,}155 MC
    \item 226 个零提前期锚点的 MC 任务降级为 skipped
      （housing 2 / POI 131 / population 93）
    \item Response Artifact：136{,}296 行，覆盖四族、方法、scope 与规格指纹
    \item 发布带输入哈希与资格凭证，标签与诊断可逐行溯源
  \end{itemize}
\end{frame}

\begin{frame}{独立聚合原则}
  \begin{itemize}
    \item 分析单元 = \textbf{family $\times$ method $\times$ donor scope}，各自独立聚合
    \item Matching：同城与跨城两个 scope 分别估计
    \item Pooled GSC/MC：同城与跨城行分别聚合，不跨 scope 混用
    \item 样本量不足以估计的组合记为 \texttt{no\_data} 并写明原因，
      不主动舍弃、不影响其他组合
    \item 本例：跨城匹配 VIIRS 13、POI 16、population 61；
      housing 无跨城 matched 标签，记 \texttt{no\_data}
  \end{itemize}
\end{frame}

    \item Pooled GSC/MC：1{,}582 个处理任务；方法与 scope 分行聚合
    \item pooled 准入拒绝 46/7{,}221：

  \end{itemize}
  \vspace{0.5em}
  \begin{block}{解读要点}
    事件研究是聚合诊断而非标签门槛；跨城与 pooled 样本有限，

\begin{frame}{结果（3）：封版与下游选择}
  \begin{itemize}
    \item 统一长表：565 行 = family $\times$ method $\times$ scope $\times$ outcome $\times$ event time
      （规范窗口内）
    \item 可用性：24 格 = 20 available + 4 \texttt{no\_data}（全部为 housing：
      跨城匹配 + pooled 缺失格），原因逐行写清
    \item 一致性校验通过：窗口内最大 $n_{\text{treated}}$ 从不超过可用性计数
    \item 下游 causal supervision 选择 \textbf{VIIRS + POI}；
      housing / population 已分析但不进入下游监督
  \end{itemize}
\end{frame}

\begin{frame}{标签获取：任务、标签定义与发布}
  \begin{itemize}
    \item 每个任务产出同一结果族、同一事件期的观测与反事实，标签
      $L=Y-\hat Y(0)$
    \item 期限：housing / VIIRS $\{1,3,6,12,18,24\}$ 个月；
      population $\{1,2,3\}$ 年；POI 四个 outcome $\times$ $\{1,2,3\}$ 年
    \item 方法：冻结匹配（\texttt{frozen\_matched\_change}）、
      GSC（\texttt{xu\_2017\_gsynth}）、MC（\texttt{athey\_2021\_mc}）
    \item 每个标签伴随：方法、标准误、可用性、质量分级与处理前拟合诊断
    \item 发布后形成 27 维响应签名；本次训练只启用 VIIRS 与 POI 的维度
  \end{itemize}
\end{frame}

\begin{frame}{权重分配：可靠性权重}
  \[
    W_i=A_i\cdot\exp(-\gamma D_i)\cdot\left(p_i^{SE}\right)^{\alpha}
  \]
  \begin{itemize}
    \item \textbf{资格 $A_i$}（硬边界）：标签不可用 / 零提前期锚点 /
      匹配 holdout 越界 $\rightarrow 0$；其余为 1
    \item \textbf{设计不可靠度 $D_i$}：可用诊断百分位的行均值
      \begin{itemize}
        \item \texttt{p\_trend}：前趋势偏离 $\sqrt{\max(\text{pre\_rmspe}^2-\text{pre\_mean\_effect}^2,0)}$
        \item \texttt{p\_holdout}：匹配 holdout 比率；\texttt{p\_validation}：MC 的 CV MSPE
        \item 分层 $(method,outcome)\to(method,family)\to method\to global$，$n\ge 30$
      \end{itemize}
    \item \textbf{精度项 $p_i^{SE}$}：$1/SE^2$ 在 $(method,outcome)$ 内
      1\%/99\% 缩尾并均值归一；匹配与无 SE 标签取 1
    \item 最终：$(method,family)$ 内缩尾 + 均值归一；只有资格失败恰为 0
  \end{itemize}
\end{frame}

\begin{frame}{表示学习（简述）}
  \begin{itemize}
    \item 目标：$f:$ 处理前特征 $\rightarrow$ 嵌入；嵌入近邻与
      \textbf{因果响应签名相似度}一致，并迁移到未见城市
    \item 监督：响应签名相关（共同观测单元上的相关）；配对权重
      $\sqrt{w_i w_j}$（设计可靠性），并按 $1/(1+SE^2)$ 收缩噪声标签
    \item 损失：InfoNCE（正样本权重 = 响应相似度）+ 嵌入距离 MSE + 多任务预测 MSE
    \item 评测：\texttt{nn\_corr@k} + 机会水平/PCA/冻结 DINOv2/外观自编码器基线，
      置换检验、bootstrap、城市留出迁移探针与响应方向 AUC
    \item 当前：正跑 W-design 主版本，随后 W0 / W-full 消融
  \end{itemize}
\end{frame}
