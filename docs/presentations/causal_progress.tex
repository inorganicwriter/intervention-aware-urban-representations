\section{估计器与事件研究}

\begin{frame}{估计器一览}
  \small
  \begin{tabular}{p{0.13\textwidth}p{0.41\textwidth}p{0.38\textwidth}}
    \toprule
    方法 & 反事实结构 & 推断 \\
    \midrule
    匹配 & M=5 候选 + 静态精炼，冻结控制身份 & placebo 门禁；完整 ATT 用 Abadie--Imbens 解析方差 \\
    GSC & 交互固定效应；目标用处理前观测拟合 & 因子数 CV；200 次参数 bootstrap \\
    MC & 核范数 + $\lambda$ 交叉验证 & 单位 jackknife（fect 等价） \\
    \bottomrule
  \end{tabular}
  \medskip
  \begin{block}{匹配质量门禁}
    \small 训练距离、holdout $RMS$、$MAX$ 均不超过 placebo 分布的 95\% 分位
  \end{block}
\end{frame}

\begin{frame}{匹配：距离、标签与完整 ATT}
  \small
  \begin{block}{Mahalanobis 距离}
    \[
      d_{ij}=\sqrt{(X_i-X_j)'\hat\Sigma^{-1}(X_i-X_j)}
    \]
  \end{block}
  \begin{block}{月度标签（$B$ 为开通前 12 个清洁月均值）}
    \[
      L_{ikh}=(Y_{i,h}-B_i)-(Y_{j(i),h}-B_{j(i)})
    \]
  \end{block}
  \begin{block}{完整趋势 ATT（Abadie--Imbens 偏差修正）}
    \[
      ATT=\frac{1}{N_T}\sum_i
      \left[\Delta Y_i-\Delta Y_{j(i)}+(X_i-X_{j(i)})'\hat\beta\right]
    \]
  \end{block}
\end{frame}

\begin{frame}{GSC 与 MC：核心公式}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{GSC：因子结构与目标拟合}
        \small
        \[ Y_{it}(0)=\alpha_i+\delta_t+\lambda_i'F_t+\varepsilon_{it} \]
        \[ Y_{i,t}=a_i+F_t'b_i+\varepsilon_{i,t} \]
        因子 $r=0,\dots,5$：CV + 1SE
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{MC：核范数与 jackknife}
        \small
        \[ \min_L\ \tfrac{1}{|\Omega|}\sum_{\Omega}(Y_{it}-L_{it})^2
        +\lambda\|L\|_* \]
        \[ SE_t^{JK}=\sqrt{\operatorname{var}(p_t^{(-j)})/N_t} \]
        $N_t$：事件期有效重拟合次数
      \end{block}
    \end{column}
  \end{columns}
  \medskip
  \begin{itemize}
    \item GSC：$\hat Y(0)=\hat a_i+\hat F_t'\hat b_i$；$SE_t=sd(\tau_t^{(b)})$
    \item MC：奇异值软阈值；$\lambda=0$ 为合法端点
  \end{itemize}
\end{frame}

\begin{frame}{事件研究与平行趋势}
  \small
  \begin{block}{TWFE 事件研究，参考期 $h_0=-1$}
    \[
      Y_{it}=\alpha_i+\gamma_t+\sum_{h\neq h_0}\beta_h
      \left[Treated_i\cdot\mathbf{1}(\text{event\_time}=h)\right]+\varepsilon_{it}
    \]
  \end{block}
  \begin{block}{规范窗口与联合检验}
    月度族 $\pm24$；年度族 $[-6,+3]$；参考期前 5 个 clean lead（$-6..-2$）
    \[
      F=q^{-1}\hat\beta_{pre}'\hat V_{pre}^{-1}\hat\beta_{pre}
      \sim F(q,\,G-1)
    \]
  \end{block}
  \begin{itemize}
    \item 聚类：CRV1（grid / city），主解释用城市聚类
    \item pre-trend 是诊断，不作为标签门槛
  \end{itemize}
\end{frame}

\section{结果}

\begin{frame}{独立聚合原则}
  \begin{itemize}
    \item 分析单元：\key{family $\times$ method $\times$ donor scope}，各自独立聚合
    \item 匹配与 pooled GSC/MC 均按 same-city / cross-city 分开
    \item 样本不足以估计的组合记 \texttt{no\_data} 并写明原因，不主动舍弃
    \item 本例：跨城匹配 VIIRS 13、POI 16、population 61；housing 记 \texttt{no\_data}
  \end{itemize}
\end{frame}

\section{标签与权重}

\begin{frame}{标签获取}
  \small
  \begin{itemize}
    \item 每任务：观测减反事实，$L=Y-\hat Y(0)$
    \item 期限：housing / VIIRS 月 $\{1,3,6,12,18,24\}$；population 年 $\{1,2,3\}$；
      POI 四个 outcome $\times$ 年 $\{1,2,3\}$
    \item 方法：冻结匹配、GSC、MC；同城与跨城 scope 独立
    \item 每个标签保存方法、标准误、可用性、质量分级与处理前拟合诊断
    \item 发布后形成 27 维响应签名；本次仅启用 VIIRS 与 POI 维度
  \end{itemize}
\end{frame}

\begin{frame}{权重生成逻辑}
  \small
  \[
    W_i=A_i\cdot\exp(-\gamma D_i)\cdot(p_i^{SE})^{\alpha}
  \]
  \begin{tabular}{p{0.2\textwidth}p{0.74\textwidth}}
    \toprule
    项 & 规则 \\
    \midrule
    资格 $A_i$ & 不可用 / 零提前期锚点 / 匹配 holdout 越界 $\rightarrow 0$；其余为 1 \\
    设计 $D_i$ & 可用诊断百分位的行均值（trend / holdout / validation），
      分层 $(method,outcome)\to(method,family)\to method\to global$，$n\ge30$ \\
    精度 $p_i^{SE}$ & $1/SE^2$ 在 $method\times outcome$ 内 1\%/99\% 缩尾并均值归一；
      匹配与无 SE 标签取 1 \\
    收尾 & $method\times family$ 内缩尾 $+$ 均值归一；仅资格失败恰为 0 \\
    \bottomrule
  \end{tabular}
  \medskip
  \begin{itemize}
    \item task sampling weight $=$ eligible 标签权重的最小值（仅供采样）
  \end{itemize}
\end{frame}

\begin{frame}{权重版本与数据集}
  \small
  \begin{tabular}{llll}
    \toprule
    版本 & 权重 & 用途 & 数据集 \\
    \midrule
    W0 & 单位（$\gamma=0,\alpha=0$） & 基线 & 待建 \\
    W-design & design-only（$\alpha=0$） & \key{主训练} & \texttt{viirs-mg-design}、\texttt{poi-mg-design} \\
    W-full & design $\times$ precision（$\alpha=0.5$） & 消融 & \texttt{*-mg-full} \\
    \bottomrule
  \end{tabular}
  \medskip
  \begin{itemize}
    \item 逐族独立：VIIRS（6 维监督）与 POI（12 维监督）各训一个模型
    \item 核验：各方法 $\times$ 族内 eligible 均值 $1$；eligible 无零、ineligible 无正值
    \item POI 四个 outcome 权重独立；构建后按 runbook 逐项记录
  \end{itemize}
\end{frame}

\section{表示学习}

\begin{frame}{表示学习（1）：目标与监督信号}
  \small
  \begin{block}{学习目标}
    $f:$ 处理前特征 $\to \mathbb{R}^d$，使嵌入近邻与
    \key{因果响应签名相似度}一致，并迁移到未见城市
  \end{block}
  \begin{itemize}
    \item 响应签名 $r_i=(L_{i,k,h})$：跨 outcome 与 horizon 的估计效应向量
    \item 相似度：共同观测单元上的加权相关；本案仅启用 VIIRS 与 POI 维度
    \item 输入只用处理前特征（lag 1--3，训练城市拟合标准化），无泄漏
  \end{itemize}
\end{frame}

\begin{frame}{表示学习（2）：模型与损失}
  \small
  \begin{itemize}
    \item 结构：表格 MLP（可选 DINOv2 街景分支）$\to$ 融合 $\to$ 128 维归一化嵌入
      $\to$ 每族预测头
    \item 配对权重 $a_{ijk}=m_{ijk}\sqrt{w_{ik}w_{jk}}$；无共同可靠 support 的 pair
      从 InfoNCE 分母剔除
  \end{itemize}
  \[
    L=(1-\lambda)\,L_{rep}+\lambda\,L_{pred},\qquad
    L_{pred}=\frac{\sum m\,w\,\ell}{\sum m\,w}
  \]
  \begin{itemize}
    \item $L_{rep}$：InfoNCE（正样本权重 $=$ 响应相似度，按 $1/(1+SE^2)$ 收缩）
      $+$ 嵌入--相似度 MSE
    \item 协议：城市级划分、\texttt{final\_training\_mask}、AdamW + cosine、
      按验证 loss 选优、多 seed
  \end{itemize}
\end{frame}

\begin{frame}{表示学习（3）：评测协议与判据}
  \small
  \begin{itemize}
    \item 检索：\texttt{nn\_corr@k}（总体 + 逐族）与随机近邻基线
    \item 对照基线：随机投影（机会）、PCA（外观几何）、冻结 DINOv2、外观自编码器
    \item 稳健性：置换检验 $p$ 值、bootstrap 区间
    \item 迁移：城市留出 \texttt{per\_city}、少样本探针曲线、交叉验证探针、
      响应方向 AUC
    \item 成功判据：显著超过全部基线、未见城市一致、跨 seed 稳定
  \end{itemize}
  \medskip
  \begin{block}{当前}
    \small W-design 主版本训练中，随后 W0 / W-full 消融；估计器开源包在规划中。
  \end{block}
\end{frame}

\begin{frame}{阶段状态与下一步}
  \small
  \begin{itemize}
    \item 已完成：四族事件研究封版（同城 + 跨城 + pooled）、严格发布、
      两个权重数据集
    \item 进行中：表示学习训练与消融（W-design / W0 / W-full）
    \item 计划：Python/GPU 估计器开源包（R parity 认证、Apache-2.0），
      作为额外 contribution
  \end{itemize}
\end{frame}
