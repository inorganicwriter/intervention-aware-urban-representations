\section{Background}

\begin{frame}{现实背景：轨道扩张与站点开发}
  \small
  \begin{itemize}
    \item 过去二十年，中国城市轨道交通快速扩张：开通城市数量与运营里程持续增长，
      多个城市进入网络化运营阶段
    \item 站点开通是典型的城市干预：同时改变可达性、土地价值、商业集聚与人口分布，
      也是 TOD 与站点综合开发政策的落点
    \item 房价挂牌、POI、夜间灯光、人口与卫星遥感等多源数据覆盖度提升，
      使站点级、多结果、跨城市的量化评估成为可能
    \item 决策需求超出"平均效应"：需要知道哪些站点、在什么条件下、对哪些结果
      响应更强
  \end{itemize}
\end{frame}

\begin{frame}{研究问题}
  \small
  \begin{itemize}
    \item 轨道交通站点开通如何改变周边：住房价格、POI（数量/熵/商业占比/交通可达）、
      夜间灯光、人口
    \item 关注动态路径：开通后逐期效应（月频 1--24，年频 1--3）
    \item 关注异质性：谁受益、受益多少、持续多久，以及不同城市之间能否比较
  \end{itemize}
  \medskip
  \begin{block}{研究意义}
    轨道投资规模大、周期长。理解响应异质性是选址、时序与配套政策的前提；
    若能学习"响应相似"的城市表示，可把已有评估迁移到未开通或待评估城市。
  \end{block}
\end{frame}

\begin{frame}{响应为什么异质}
  \small
  \begin{itemize}
    \item 可达性改善：通勤时间、换乘便利、就业与活动可达范围
    \item 集聚与商业重组：新商业进入、既有商业挤出、业态结构变化
    \item 分流与替代：与既有站点、公交与道路网络的竞争关系
    \item 预期与资本化：开通前的价格预期、开发时序与土地供给响应
    \item 调节变量：站点类型（新线/延伸/换乘/终点）、网络位置、城市规模、
      周边人口与配套
  \end{itemize}
  \medskip
  \begin{block}{含义}
    同一干预在不同地点可正可负、可快可慢，单一平均效应不足以支持选址与配套决策。
  \end{block}
\end{frame}

\begin{frame}{从单案例到跨城市}
  \small
  \begin{itemize}
    \item 传统评估回答"这个城市或线路发生了什么"，结论常受本地条件约束
    \item 规划与投资决策更关心"哪些地方与已知案例相似、可能产生类似响应"
    \item 这要求把每个事件的响应路径变成可比较、可检索、可迁移的对象，
      而不只是若干回归系数
    \item 迁移的三个层次：同结果跨城市、同城市跨结果、新城市零样本或少样本
  \end{itemize}
\end{frame}

\begin{frame}{主要困难}
  \small
  \begin{itemize}
    \item 无实验：每城市开通站点有限、时点不同，选址与开通时点内生
    \item 多期多结果：单一 DiD 系数刻画不了完整响应路径，更无法跨结果比较
    \item 样本异质：城市规模、网络形态、站点类型差异大，处理前特征维度高
    \item 逐案重跑：每评估一个城市或线路都要重新构造对照与估计
    \item 缺少可迁移表示：既有研究多为单城市或单线路结论，难以直接外推
  \end{itemize}
\end{frame}

\begin{frame}{传统工具与局限}
  \small
  \begin{tabular}{p{0.24\textwidth}p{0.34\textwidth}p{0.34\textwidth}}
    \toprule
    工具 & 作用 & 局限 \\
    \midrule
    DiD / TWFE 事件研究 & 动态处理效应、可加固定效应 & 平行趋势敏感；逐案重估 \\
    合成控制（SCM） & 单处理单元的反事实路径 & 一次一个处理单元，难规模化 \\
    广义合成控制（GSC） & 交互固定效应、可处理多个处理单元 & 依赖线性因子结构 \\
    矩阵补全 & 面板反事实填充、不确定性完整 & 结果逐网格独立，不做响应相似性 \\
    匹配/合成对照 & 控制处理前差异 & 匹配协变量与设计诊断决定可信度 \\
    \bottomrule
  \end{tabular}
  \medskip
  \begin{block}{现有工具的不足}
    这些工具产出逐任务的响应估计，但不产出"任务之间响应是否相似"的可迁移表示，
    也没有把估计质量显式带入下游学习。
  \end{block}
\end{frame}

\begin{frame}{项目核心思路}
  \small
  \begin{block}{把因果估计本身当作监督}
    对每个处理事件 $i$ 得到响应路径 $r_i$（跨 horizon 与结果的效应向量）；学习
    $f:\text{处理前特征}\to\mathbb{R}^d$，使
    $\mathrm{sim}_{\text{emb}}(f_i,f_j)$ 与 $\mathrm{sim}_{\text{resp}}(r_i,r_j)$ 一致。
  \end{block}
  \begin{itemize}
    \item 标签不是人工标注，而是设计型因果估计：matched 事件研究、GSC、矩阵补全
    \item 估计质量显式进入下游：预趋势、holdout、CV MSPE 等诊断决定监督资格与
      逐标签可靠性权重
    \item 产出两层资产：可审计的因果响应标签库与响应对齐的城市表示
  \end{itemize}
\end{frame}

\begin{frame}{从政策问题到学习任务}
  \small
  \begin{itemize}
    \item 事件 $i$：处理前特征 $x_i$（多模态，lag 1--3）、响应路径 $r_i$
      （跨 horizon 与结果的效应向量）
    \item 目标：学习 $f$，使嵌入相似度与响应相似度一致，并迁移到训练未见城市
    \item 标签来自设计型因果估计而非人工标注；识别诊断决定标签资格与可靠性权重
    \item 两个评价维度：检索（近邻是否共享响应）与迁移（新城市的方向、幅度、
      形状预测）
  \end{itemize}
\end{frame}

\begin{frame}{项目目标与贡献}
  \small
  \begin{itemize}
    \item 资产：覆盖 5{,}048 个站点事件的响应标签库（多估计量、多结果、可审计）
    \item 方法：因果响应监督的表示学习框架（软标签对比、预测头、可靠性权重）
    \item 评估：以响应几何为准的检索与迁移协议，含外观/机会基线与城市级交叉验证
    \item 边界：系统检验因果监督何时有效、何时失效，给出可迁移性边界
  \end{itemize}
\end{frame}

\section{Related Work}

\begin{frame}{相关文献（1）：面板因果估计}
  \small
  \begin{itemize}
    \item 双重差分与事件研究：Callaway \& Sant'Anna (2021)；Sun \& Abraham (2021)；
      Goodman-Bacon (2021)
    \item 合成控制与广义合成控制：Abadie \& Gardeazabal (2003)；Abadie, Diamond
      \& Hainmueller (2010)；Xu (2017)
    \item 矩阵补全：Athey, Bayati, Doudchenko, Imbens \& Khosravi (2021)
    \item 匹配/合成对照与诊断：Imai, Kim \& Wang (2023)；Roth (2022)；
      Rambachan \& Roth (2023)
  \end{itemize}
  \medskip
  \begin{block}{本项目的做法}
    不发明新的估计量：在同一批事件上并行运行三种估计量，用识别诊断决定哪些标签
    进入监督、以多大权重进入，并把发布冻结成可复现资产。
  \end{block}
\end{frame}

\begin{frame}{相关文献（2）：交通基础设施影响评估}
  \small
  \begin{itemize}
    \item 铁路/轨道开通的局部影响：Gibbons \& Machin (2005)；
      Gonzalez-Navarro \& Turner (2018)
    \item 中国城市轨道与土地利用/房价：Zheng \& Kahn (2013) 等
    \item 共同点：以个别城市或线路为对象，报告若干结果的平均或动态效应
  \end{itemize}
  \medskip
  \begin{block}{现有文献的不足}
    缺少跨城市、多结果、动态路径的系统性标签库，也缺少把响应形状作为可检索、
    可迁移对象的表示学习框架。
  \end{block}
\end{frame}

\begin{frame}{相关文献（3）：城市表示学习与遥感表征}
  \small
  \begin{itemize}
    \item 对比/自监督表示：SimCLR（Chen et al., 2020）；MoCo（He et al., 2020）；
      SupCon（Khosla et al., 2020）；DINOv2（Oquab et al., 2024）
    \item 卫星影像与城市/贫困制图：Jean et al. (2016)；MOSAIKS（Rolf et al., 2021）
    \item 监督信号通常来自图像增强、任务标签或下游指标
  \end{itemize}
  \medskip
  \begin{block}{与本项目的区别}
    本项目使用因果估计的响应路径作为软标签监督，目标是让嵌入几何与干预响应几何
    一致，而不是图像或外观相似，也不是单一任务精度。
  \end{block}
\end{frame}

\begin{frame}{相关文献（4）：与因果表示学习的边界}
  \small
  \begin{itemize}
    \item 因果表示学习（CRL，Sch\"olkopf et al., 2021）的目标是从观测中学习
      潜在因果变量与结构
    \item 本项目不是 CRL：因果结构由设计型估计流程给定并冻结，表示学习只负责把
      响应相似性迁移到未见城市
    \item 与因果监督的预测模型的区别：不预测单个效应，而是学习可用于检索、比较
      与迁移的嵌入空间
  \end{itemize}
\end{frame}

\begin{frame}{定位对照}
  \small
  \begin{tabular}{p{0.26\textwidth}p{0.31\textwidth}p{0.35\textwidth}}

\section{方法与资产}

\begin{frame}{数据与事件}
  \small
  \begin{itemize}
    \item 冻结处理清单：5{,}048 个站点开通事件（城市、网络、开通月份）
    \item 结果族与 horizon：住房价格与夜间灯光（月频 1--24）；POI 四指标与人口
      （年频 1--3）
    \item 处理前特征：房价、POI、灯光、人口、Sentinel-2 的 1--3 年 lag；
      标准化参数只用训练城市拟合
    \item 可选输入：街景图像（冻结 DINOv2 编码器），默认关闭
  \end{itemize}

\begin{frame}{表示学习：模型与目标}
  \small
  \begin{itemize}
    \item 结构：表格 MLP（可选街景分支）$\to$ 融合 $\to$ 128 维归一化嵌入
      $\to$ 每结果族预测头
    \item 软标签对比：正样本权重等于响应相似度（逐族相关、可权重化）；
      无共同可靠 support 的 pair 从分母剔除
    \item 预测头：逐 cell 加权 MSE（设计权重）；表示损失与预测损失加权合成
    \item 监督口径：逐结果族独立（VIiRS 6 维 / POI 12 维），只用同城标签
  \end{itemize}
\end{frame}

\begin{frame}{评估协议}
  \small
  \begin{itemize}
    \item 检索：近邻响应相似度 \texttt{nn\_corr@k}；含随机近邻基线
    \item 对照：原始特征检索、PCA、随机投影、冻结 DINOv2、外观自编码器
    \item 稳健性：置换检验、bootstrap 区间、多 seed
    \item 迁移：逐城市检索、少样本探针、方向预测 AUC
    \item 诊断补充：城市级交叉验证（折内隔离预处理与公平基线）、方向/幅度/形状
      分开报告
  \end{itemize}
\end{frame}

\section{现状与发现}

\begin{frame}{评估修正与诊断}
  \small
  \begin{itemize}
    \item 评估流程经三轮独立审阅：probe 含截距与均值基线、折内预处理隔离、
      公平 refit 基线、条件化规则版本化、城市网格数与 cell 数分离
    \item 已排除：标签噪声（VIiRS 折半信度约 0.8）、两点支撑问题、float32 消减、
      整体嵌入塌缩
    \item 已确认：设计权重在 horizon 恒定，W-design 不重塑成对相似度；
      对比损失与距离损失存在几何目标冲突（待验证机制）
    \item 未决：原始特征的城市外可预测性
  \end{itemize}
\end{frame}
